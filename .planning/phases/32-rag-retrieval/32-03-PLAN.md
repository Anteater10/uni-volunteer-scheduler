    Implementation (`rerank.py`):
    ```python
    from functools import lru_cache
    from sentence_transformers import CrossEncoder

    @lru_cache(maxsize=1)
    def _model() -> CrossEncoder:
        return CrossEncoder("BAAI/bge-reranker-base", max_length=512)

    def rerank(query: str, candidates: list[dict], top_k: int = 5) -> list[dict]:
        if not candidates:
            return []
        pairs = [(query, c["content"]) for c in candidates]
        scores = _model().predict(pairs, batch_size=16, show_progress_bar=False)
        # tiebreak by id ascending
        scored = sorted(
            zip(candidates, scores),
            key=lambda cs: (-float(cs[1]), str(cs[0]["id"])),
        )
        # attach rerank_score for downstream
        out = []
        for c, s in scored[:top_k]:
            out.append({**c, "rerank_score": float(s)})
        return out
    ```
