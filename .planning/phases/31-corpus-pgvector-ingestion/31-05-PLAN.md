       """REQ-31-12 — EXPLAIN over a cosine query uses the HNSW index."""
       import pytest
       from sqlalchemy import text


       def test_hnsw_index_used(alembic_engine, fake_embedding_provider, tiny_markdown_corpus, db_session):
           """After ingest + build-index, EXPLAIN over `ORDER BY embedding <=> ...` shows HNSW scan."""
           from app.corpus.ingest import run_ingestion, build_hnsw_index
           run_ingestion(root=tiny_markdown_corpus, provider=fake_embedding_provider, session=db_session)
           build_hnsw_index(session=db_session)

           db_session.execute(text("SET LOCAL enable_seqscan = off"))

               EXPLAIN SELECT id FROM corpus_chunks
               ORDER BY embedding &lt;=&gt; (SELECT embedding FROM corpus_chunks LIMIT 1)
               LIMIT 5
           """)).all()
