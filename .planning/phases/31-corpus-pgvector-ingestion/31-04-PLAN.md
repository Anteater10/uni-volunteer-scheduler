<objective>
Compose the chunker + walker + a pluggable embedder into an idempotent ingestion CLI. After this plan, `python -m app.corpus.ingest --source docs --commit` walks the project docs, embeds them via Jina (or falls back to local BGE), and produces real rows in `corpus_documents`, `corpus_chunks`, and `ingestion_runs`. The HNSW index is built post-ingest via `--build-index`.

Purpose: this is the load-bearing plan of Phase 31. Everything from plans 01-03 was substrate; this is where the corpus actually materialises. The idempotency contract (content_sha256-keyed upsert) is what makes the paper's reproducibility claim survive — Phase 35 can re-derive any ingestion run from `(git_commit_sha, source_globs, chunker_version)`.

```python
from typing import Protocol, NamedTuple

EMBEDDING_DIM = 1024  # locked, matches Vector(1024) on corpus_chunks

class EmbedMeta(NamedTuple):
    provider: str         # 'jina' | 'local-bge'
    model_id: str         # 'jina-embeddings-v3' | 'BAAI/bge-small-en-v1.5+pad1024'
    api_calls: int
    latency_ms: int
    tokens: int           # 0 for local; provider-reported for Jina

class EmbeddingProvider(Protocol):
    name: str
    model_id: str
    def embed(self, texts: list[str]) -> tuple[list[list[float]], EmbedMeta]: ...

class JinaEmbeddingProvider:
    def __init__(self, api_key: str, model: str = "jina-embeddings-v3"): ...
    def embed(self, texts: list[str]) -> tuple[list[list[float]], EmbedMeta]: ...
    # Raises RateLimitError on 429 so ingest.py can fall back.

class LocalBgeEmbeddingProvider:
    def __init__(self, model: str = "BAAI/bge-small-en-v1.5"): ...
    def embed(self, texts: list[str]) -> tuple[list[list[float]], EmbedMeta]: ...
    # Pads native 384-dim to 1024 with zeros.
```

       **Top of run:**
       a. Capture `git_commit_sha = subprocess.check_output(["git", "rev-parse", "HEAD"]).strip().decode()`; capture `git_dirty = subprocess.check_output(["git", "status", "--porcelain"]).strip() != b""`.

       **Per document loop:**
       a. For each `SourceDocument` from `walk_sources(root=root)`:
          - Compute `content_sha256 = hashlib.sha256(doc.content.encode()).hexdigest()`.
          - SELECT existing row WHERE `source_path = doc.source_path AND content_sha256 = doc.content_sha256`. If present → `files_unchanged += 1`, skip.
          - Otherwise, in a per-document transaction:
            * INSERT `CorpusDocument(...)`.
            * `chunks = chunk_text(doc.content, chunk_size=settings.corpus_chunk_size, chunk_overlap=settings.corpus_chunk_overlap)`.
            * Embed in batch via `provider.embed([c.content for c in chunks])`; on `RateLimitError` AND `fallback_provider` is set, switch to fallback (record `notes += f"fell back to {fallback.name}"`) and retry the batch ONCE.
            * INSERT each `CorpusChunk(...)` with the embedding vector, `embedding_provider`/`embedding_model` set per the provider that produced it.
            * Increment counters: `files_ingested += 1`, `chunks_emitted += len(chunks)`, `chunks_embedded += len(chunks)`, `embedding_api_calls += meta.api_calls`, `embedding_latency_ms_total += meta.latency_ms`, `embedding_tokens_total += meta.tokens`.
            * Commit the per-document transaction.
          - On any exception in the per-doc block: rollback that transaction, increment `files_failed`, log `error_class`/`error_message` on the run row, continue to next document (do not abort).

       **End of run:**
       - UPDATE `ingestion_runs` row: set `completed_at = now()`. `status='succeeded'` if `files_failed == 0` else `'failed'` (or `'partial'` if some succeeded and some failed — RESEARCH §Step 2 enumerates `'running' | 'succeeded' | 'partial' | 'failed'`).
       - Commit.
       - Return `IngestionResult(...)`.

       ```python
       """CLI entry for corpus ingestion.

       Canonical invocation (inside compose network):
         docker compose run --rm backend python -m app.corpus.ingest --source docs --commit
       """
       from __future__ import annotations
       import argparse
       import json
       import sys
       from pathlib import Path

       from app.corpus.ingest import run_ingestion, build_hnsw_index
       from app.corpus.embeddings import (
           JinaEmbeddingProvider,
           LocalBgeEmbeddingProvider,
           get_primary_and_fallback,
       )
       from app.database import SessionLocal
       from app.config import settings


       def main(argv: list[str] | None = None) -> int:
           p = argparse.ArgumentParser(prog="python -m app.corpus.ingest")
           p.add_argument("--source", type=Path, default=Path("."), help="root directory to walk")
           mode = p.add_mutually_exclusive_group()
           mode.add_argument("--commit", action="store_true", help="write to DB (default if neither --commit nor --dry-run)")
           mode.add_argument("--dry-run", action="store_true", help="walk + chunk + hash, no DB writes")
           p.add_argument("--provider", choices=["jina", "local"], default=settings.corpus_embedding_primary)
           p.add_argument("--rebuild", action="store_true", help="TRUNCATE corpus_chunks + corpus_documents first")
           p.add_argument("--build-index", action="store_true", help="CREATE INDEX … USING hnsw (idempotent); skips ingestion")
           args = p.parse_args(argv)

           session = SessionLocal()
           try:
               if args.build_index:
                   build_hnsw_index(session=session)
                   print("HNSW index ensured.")
                   return 0

               if args.provider == "jina":
                   primary = JinaEmbeddingProvider(api_key=settings.jina_api_key, model=settings.jina_embedding_model)
                   fallback = LocalBgeEmbeddingProvider(model=settings.local_embedding_model)
               else:
                   primary = LocalBgeEmbeddingProvider(model=settings.local_embedding_model)
                   fallback = None

               result = run_ingestion(
                   root=args.source,
                   provider=primary,
                   fallback_provider=fallback,
                   session=session,
                   dry_run=args.dry_run,
                   rebuild=args.rebuild,
               )
               print(json.dumps(result.__dict__, indent=2, default=str))
               return 0 if result.status in ("succeeded", "partial") else 1
           finally:
               session.close()


       if __name__ == "__main__":
           sys.exit(main())
       ```
